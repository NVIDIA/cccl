#include <cuda/std/cstdint>

#include <cub/block/block_load.cuh>

using block_load_t =
    cub::BlockLoad<::cuda::std::int32_t, 128, 2,
                   ::cub::BLOCK_LOAD_TRANSPOSE, 1, 1>;

using load_storage_t =
    typename block_load_t::TempStorage;

__device__ constexpr unsigned load_storage_bytes =
    sizeof(load_storage_t);

__device__ constexpr unsigned load_storage_alignment =
    alignof(load_storage_t);

extern "C" __device__ void
load_impl(load_storage_t *temp_storage,
          ::cuda::std::int32_t *param_0,
          ::cuda::std::int32_t *param_1) {
  block_load_t(*temp_storage)
      .Load(param_0, *reinterpret_cast<
                         ::cuda::std::int32_t (*)[2]>(
                         param_1));
}

extern "C" __device__ int
load_abi(void *__ret, void *abi_param_0,
         void *abi_param_1, void *abi_param_2) {
  load_storage_t *abi_cast_0 =
      reinterpret_cast<load_storage_t *>(abi_param_0);
  ::cuda::std::int32_t *abi_cast_1 =
      reinterpret_cast<::cuda::std::int32_t *>(
          abi_param_1);
  ::cuda::std::int32_t *abi_cast_2 =
      reinterpret_cast<::cuda::std::int32_t *>(
          abi_param_2);
  load_impl(abi_cast_0, abi_cast_1, abi_cast_2);
  return 0;
}
