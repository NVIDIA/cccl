#include <cuda/std/cstdint>

#include <cub/block/block_load.cuh>

using block_load_t =
    cub::BlockLoad<::cuda::std::int32_t, 128, 2,
                   ::cub::BLOCK_LOAD_TRANSPOSE, 1, 1>;

using load_storage_t =
    typename block_load_t::TempStorage;

__device__ constexpr unsigned load_storage_bytes =
    sizeof(load_storage_t);

__device__ constexpr unsigned load_storage_alignment =
    alignof(load_storage_t);

extern "C" __device__ void
load_impl(load_storage_t *temp_storage,
          ::cuda::std::int32_t *src,
          ::cuda::std::int32_t *dst) {
  block_load_t(*temp_storage)
      .Load(src,
            *reinterpret_cast<
                ::cuda::std::int32_t (*)[2]>(dst));
}

extern "C" __device__ int load_abi(void *__ret,
                                   void *temp_storage,
                                   void *src,
                                   void *dst) {
  load_storage_t *cast_temp_storage =
      reinterpret_cast<load_storage_t *>(
          temp_storage);
  ::cuda::std::int32_t *cast_src =
      reinterpret_cast<::cuda::std::int32_t *>(src);
  ::cuda::std::int32_t *cast_dst =
      reinterpret_cast<::cuda::std::int32_t *>(dst);
  load_impl(cast_temp_storage, cast_src, cast_dst);
  return 0;
}
