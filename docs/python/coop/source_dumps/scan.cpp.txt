#include <cuda/std/cstdint>

#include <cub/block/block_scan.cuh>

using block_scan_t =
    cub::BlockScan<::cuda::std::int32_t, 128,
                   ::cub::BLOCK_SCAN_RAKING, 1, 1>;

using scan_storage_t =
    typename block_scan_t::TempStorage;

__device__ constexpr unsigned scan_storage_bytes =
    sizeof(scan_storage_t);

__device__ constexpr unsigned scan_storage_alignment =
    alignof(scan_storage_t);

extern "C" __device__ ::cuda::std::int32_t
maximum_device(::cuda::std::int32_t,
               ::cuda::std::int32_t);

extern "C" __device__ void
scan_impl(scan_storage_t *temp_storage,
          ::cuda::std::int32_t &input,
          ::cuda::std::int32_t &output) {
  auto scan_op =
      [](const ::cuda::std::int32_t &wp_0,
         const ::cuda::std::int32_t &wp_1) {
        return maximum_device(wp_0, wp_1);
      };
  block_scan_t(*temp_storage)
      .InclusiveScan(input, output, scan_op);
}

extern "C" __device__ int
scan_abi(void *__ret, void *temp_storage,
         ::cuda::std::int32_t input) {
  scan_storage_t *cast_temp_storage =
      reinterpret_cast<scan_storage_t *>(
          temp_storage);
  ::cuda::std::int32_t ref_input = input;
  ::cuda::std::int32_t out;
  scan_impl(cast_temp_storage, ref_input, out);
  *reinterpret_cast<::cuda::std::int32_t *>(__ret) =
      out;
  return 0;
}
